\begin{frame}{Financing conditions matter for the energy transition}
\lead{Wind and solar concentrate spending at the start of a project.}
\begin{columns}[T,onlytextwidth]
\column{0.55\textwidth}
\begin{itemize}
\item Debt and equity providers require a return on the capital they commit.
\item A higher financing cost raises the revenue a project must earn.
\item Regional financing differences can change modeled transition pathways.
\end{itemize}
\column{0.40\textwidth}
\vspace{0.35cm}
{\color{DukeRoyalBlue}\Large WACC}\\[0.3cm]
\small Weighted average cost of capital
\[
\underbrace{Lr_D(1-\tau)}_{\text{debt}}+
\underbrace{(1-L)r_E}_{\text{equity}}
\]
\end{columns}
\source{Ameli et al., Nature Communications (2021); Calcaterra et al., Nature Energy (2024).}
\end{frame}

\begin{frame}{Fuel-price forecasting combines temporal learning and smoothing}
\begin{columns}[T,onlytextwidth]
\column{0.53\textwidth}
\lead{Sequence model}
\begin{itemize}
\item A 12-month input window captures recent price dynamics.
\item Seasonal sine/cosine features represent the month of year.
\item CNN, recurrent layers, and attention learn temporal representations.
\end{itemize}
\column{0.42\textwidth}
\lead{Gaussian process}
\begin{itemize}
\item An RBF kernel smooths the neural forecast path.
\item The resulting path feeds the WACC comparison.
\end{itemize}
\smalllabel{Its conditional variance does not represent the full uncertainty of the neural forecasting stage.}
\end{columns}
\source{Lindemann et al. (2021); Lahmiri (2024); Rasmussen (2004). Implementation settings are provided in the paper appendix.}
\end{frame}

\begin{frame}{WACC regressions and chronological checks serve different roles}
\begin{columns}[T,onlytextwidth]
\column{0.47\textwidth}
\lead{Original AutoGluon fits}
\[
\mathrm{WACC}_{r,p,t}=f_{r,p,e}(P_{e,t},\pi_{r,t})
\]
\begin{itemize}
\item One fuel price and an inflation covariate
\item Bagging and stacking
\item In-sample $R^2$, MAPE, and RMSE
\end{itemize}
\column{0.48\textwidth}
\lead{Additional diagnostics}
\begin{itemize}
\item OLS/Ridge with cleaned regional inputs
\item Train 2026--2032; test 2033--2035
\item Rolling one-year-ahead checks
\item Compare fuel + inflation + trend against a time trend
\end{itemize}
\end{columns}
\source{Fit describes the modeled relationship. Chronological tests measure how that relationship transfers to later scenario years.}
\end{frame}
